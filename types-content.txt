export interface LearningPathStep {
  step: number;
  title: string;
  description: string;
  skillUnlocked: string;
  courseId?: string;
  estimatedHours: number;
  isUnlocked: boolean;
}

export interface PracticeQuestionItem {
  id: string;

  skill: string;

export type AIMessage =
  | {
      id: string;
      sender: 'user' | 'assistant';
      type: 'text';
      content: string;
      timestamp: string;
    }
  | {
      id: string;
      sender: 'assistant';
      type: 'learning_path';
      content: string;
      steps: LearningPathStep[];
      timestamp: string;
    }
  | {
      id: string;
      sender: 'assistant';
      type: 'practice_questions';
      content: string;
      questions: PracticeQuestionItem[];
      timestamp: string;
    }
  | {
      id: string;
      sender: 'assistant';
      type: 'skill_analysis';
      content: string;
      strengths: SkillAnalysisBar[];
      weaknesses: SkillAnalysisBar[];
      timestamp: string;
    };

export interface QuickPrompt {

import type { Skill, SkillLevel } from './skills';

export type QuestionType =
  | 'MCQ'
  | 'MULTI_SELECT'
  | 'TRUE_FALSE'
  | 'CODING'
  | 'SCENARIO';

export interface Question {
  id: string;
  assessmentId: string;
  prompt: string;
  type: QuestionType;
  options: string[];
  correctAnswers: number[]; // Indices of correct options
  explanation: string;
  codeSnippet?: string;
}

export interface Assessment {
  id: string;
  title: string;
  skillId: string;
  skill: Skill;
  targetLevel: SkillLevel;
  questionCount: number;
  timeLimitMinutes: number;
  passingScorePercentage: number; // e.g., 70
  previousScore?: number;
  previousPassed?: boolean;
  questions?: Question[];
}

export interface AssessmentAttempt {
  assessmentId: string;
  startedAt: string;
  timeRemainingSeconds: number;
  answers: Record<string, number[]>; // questionId -> selectedOptionIndices
  flaggedQuestionIds: string[];
}

export interface AssessmentQuestionReview {
  question: Question;
  selectedAnswers: number[];
  isCorrect: boolean;
}

export interface AssessmentResult {
  id: string;
  assessmentId: string;
  assessmentTitle: string;
  skill: Skill;
  score: number;
  totalQuestions: number;
  percentage: number;
  passed: boolean;
  passingScorePercentage: number;
  timeTakenSeconds: number;
  completedAt: string;
  previousSkillLevel: SkillLevel | null;
  newVerifiedLevel: SkillLevel;
  newVerificationStatus: 'ASSESSED';
  nextRecommendedStep: string;
  questionReviews: AssessmentQuestionReview[];
}

export type SkillLevel = 'Beginner' | 'Intermediate' | 'Advanced' | 'Expert';

export type VerificationStatus =
  | 'SELF_DECLARED'
  | 'ASSESSED'
  | 'PROJECT_VERIFIED'
  | 'VERIFIED';

export type SkillCategory =
  | 'Programming'
  | 'Backend'
  | 'Database'
  | 'DevOps'
  | 'Cloud'
  | 'Testing'
  | 'Architecture';

export interface Skill {
  id: string;
  name: string;
  category: SkillCategory;
  description: string;
}

export interface UserSkill {
  id: string;
  userId: string;
  skillId: string;
  skill: Skill;
  level: SkillLevel;
  verificationStatus: VerificationStatus;
  expiresAt?: string;
  verifiedAt?: string;
  assessedScore?: number;
}

export type SkillGapSeverity = 'CRITICAL' | 'SECONDARY';
export type GapMagnitude = 'NONE' | 'MEDIUM' | 'HIGH';

export interface SkillGap {
  skill: Skill;
  current: SkillLevel | null;
  required: SkillLevel;
  gap: GapMagnitude;
  severity: SkillGapSeverity;
  roadmapCourseId?: string;
  roadmapPhaseId?: string;
}

export interface CareerRequirement {
  skillId: string;
  skill: Skill;
  requiredLevel: SkillLevel;
  importance: 'CRITICAL' | 'SECONDARY' | 'OPTIONAL';
  weightPercentage?: number;
}

export interface CareerRole {
  id: string;
  title: string;
  description: string;
  category: string;
  targetScore: number;
  requirements: CareerRequirement[];
  averageSalary?: string;
}

export interface CareerMatch {
  careerId: string;
  role: CareerRole;
  matchPercentage: number;
  readinessLevel: string;
  criticalGapsCount: number;
  secondaryGapsCount: number;
}
